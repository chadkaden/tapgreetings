Linger & Loop Hold (0:07.0–0:08.0s): Floating accents settle into a loop-ready resting pose, displaying the clean URL slug on the bottom frame.

1. Aspect Ratio & Crop Mathematics
A 9:16 vertical render (
1080×1920 px) has an aspect ratio of 
0.5625. Standard CR80 credit cards require:

Physical Trimmed Ratio (
2.125"×3.375"): 
0.6296 (approx. 
1:1.588).

Physical Bleed Ratio (
2.250"×3.500" or 
2.375"×3.625"): 
0.6428 to 
0.6551 (
638×1013 px @ 300 DPI).

Web-Ready CR80 Ratio (
1080 px base): Height must be trimmed from 
1920 px down to 
1715 px (
1080× 
2.125
3.375
​
 ).

By center-cropping 
102 px from the top and 
103 px from the bottom, the 9:16 image converts to a credit card ratio with zero side compression or face distortion.

2. Automated Image Pre-Cropper (process_assets.js)
This script processes your raw downloads folder, crops the images to CR80 format using sharp, and prepares clean exports for both Shopify and TapGreetings:

JavaScript
// npm install sharp
const fs = require('fs');
const path = require('path');
const sharp = require('sharp');

const DIRS = {
  raw: path.join(__dirname, 'raw_downloads'),
  cardsWeb: path.join(__dirname, 'dist', 'cards_cr80_web'),
  cardsPrint: path.join(__dirname, 'dist', 'cards_cr80_print'),
  videos: path.join(__dirname, 'dist', 'videos_9x16')
};

// Ensure directories exist
Object.values(DIRS).forEach(dir => fs.mkdirSync(dir, { recursive: true }));

async function processCardAsset(cardId, rawImageFilename, rawVideoFilename) {
  const rawImagePath = path.join(DIRS.raw, rawImageFilename);
  const rawVideoPath = path.join(DIRS.raw, rawVideoFilename);

  // 1. Move/Organize Video (Leaves 9:16 untouched for mobile playback)
  if (fs.existsSync(rawVideoPath)) {
    const videoDest = path.join(DIRS.videos, `${cardId}.mp4`);
    fs.copyFileSync(rawVideoPath, videoDest);
    console.log(`✓ Video staged: ${videoDest}`);
  }

  if (!fs.existsSync(rawImagePath)) {
    console.error(`✗ Image not found: ${rawImagePath}`);
    return;
  }

  // 2. Web-Ready CR80 Crop (1080 x 1715 px - exact 2.125" x 3.375" ratio)
  const webCardPath = path.join(DIRS.cardsWeb, `${cardId}_card_web.png`);
  await sharp(rawImagePath)
    .resize({
      width: 1080,
      height: 1715,
      fit: 'cover',
      position: 'center'
    })
    .png({ quality: 95 })
    .toFile(webCardPath);

  // 3. Print-Ready 300 DPI CR80 with Bleed (638 x 1013 px)
  const printCardPath = path.join(DIRS.cardsPrint, `${cardId}_print_cr80.png`);
  await sharp(rawImagePath)
    .resize({
      width: 638,
      height: 1013,
      fit: 'cover',
      position: 'center'
    })
    .withMetadata({ density: 300 })
    .png()
    .toFile(printCardPath);

  console.log(`✓ CR80 Card Cropped: ${webCardPath}`);
  console.log(`✓ 300 DPI Print Ready: ${printCardPath}`);
}

// Example execution for Thanksgiving & Christmas
(async () => {
  await processCardAsset('TG-01', 'TG01_flow_raw.png', 'TG01_flow_raw.mp4');
  await processCardAsset('TG-03', 'TG03_flow_raw.png', 'TG03_flow_raw.mp4');
})();

5. Automated Asset Pre-Cropper (scripts/process_assets.js)
This Node.js script ingests raw 9:16 or 1:1 Flow renders, applies non-distorting center crops for CR80 card targets, and verifies 300 DPI print metadata[cite: 1, 2]:

JavaScript
/**
 * scripts/process_assets.js
 * Ingests raw Flow visual downloads, crops them to CR80 ratios, and produces:
 * 1. Print-Ready 300 DPI image (638 x 1013 px with 0.125" bleed)[cite: 1, 2]
 * 2. Web-Ready CR80 image (1080 x 1715 px)
 * 3. Stages 9:16 MP4 video untouched for mobile gyro playback[cite: 1]
 */

const fs = require('fs');
const path = require('path');
const sharp = require('sharp');
const masterData = require('../config/tapgreetings_master.json');

const PATHS = {
  raw: path.join(__dirname, '../assets/raw_downloads'),
  artFull: path.join(__dirname, '../assets/art_full'),
  artWeb: path.join(__dirname, '../assets/art_web'),
  videos: path.join(__dirname, '../assets/videos')
};

// Ensure directories exist
Object.values(PATHS).forEach((dir) => fs.mkdirSync(dir, { recursive: true }));

async function processCard(cardId) {
  const card = masterData.cards[cardId];
  if (!card) throw new Error(`Card ID ${cardId} not found in master data`);

  const rawImage = path.join(PATHS.raw, `${cardId}_raw.png`);
  const rawVideo = path.join(PATHS.raw, `${cardId}_raw.mp4`);

  // 1. Process Video Asset
  if (fs.existsSync(rawVideo)) {
    const videoDest = path.join(PATHS.videos, `${card.taxonomy.category_folder}/${cardId}.mp4`);
    fs.mkdirSync(path.dirname(videoDest), { recursive: true });
    fs.copyFileSync(rawVideo, videoDest);
    console.log(`[Video] Staged 9:16 video to: ${videoDest}`);
  }

  // 2. Process Image Assets
  if (!fs.existsSync(rawImage)) {
    console.warn(`[Image] Raw image missing: ${rawImage}`);
    return;
  }

  // A. 300 DPI Print File (638 x 1013 px with bleed)[cite: 1, 2]
  const printDest = path.join(PATHS.artFull, `${cardId}_${card.title.replace(/\s+/g, '')}_300dpi.png`);
  await sharp(rawImage)
    .resize({
      width: 638,
      height: 1013,
      fit: 'cover',
      position: 'center'
    })
    .withMetadata({ density: 300 })
    .png({ quality: 100 })
    .toFile(printDest);
  console.log(`[Print] Generated 300 DPI CR80 image: ${printDest}`);

  // B. Web Card Display Crop (1080 x 1715 px)
  const webDest = path.join(PATHS.artWeb, `${cardId}_${card.title.replace(/\s+/g, '')}_CR80.png`);
  await sharp(rawImage)
    .resize({
      width: 1080,
      height: 1715,
      fit: 'cover',
      position: 'center'
    })
    .png({ quality: 95 })
    .toFile(webDest);
  console.log(`[Web] Generated CR80 preview image: ${webDest}`);
}

(async () => {
  for (const id of Object.keys(masterData.cards)) {
    await processCard(id);
  }
})();
